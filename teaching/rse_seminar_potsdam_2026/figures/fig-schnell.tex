% Empirical research process in the social sciences -- from
% pub_rse_methodology/schnell2_uml.tex (after Schnell et al. 1999).
% The RSE-cycle icons mark the stages where research software enters.
\documentclass[tikz,border=6pt]{standalone}
\input{tikz-preamble.tex}
\begin{document}
    \begin{tikzpicture}[]

    \node[initial] (start) {};
    \node[action, right=of start] (problem) {Forschungsproblem\\wählen};
    \node[action, right=of problem] (theory) {Theorie bilden};

    \node[fork, below=of theory] (fork1) {};

    \node[action, below left=of fork1] (concept) {Konzept spezifizieren /\\Operationalisierung};
    \node[fork, below=of concept, yshift=-4mm] (merge_concept) {};
    \node[action, below=of merge_concept] (simulation) {Simulation erstellen};

    \node[action, below=of fork1] (form) {Studiendesign\\festlegen};
    \node[action, below=of form] (unit) {Analyseeinheit\\wählen};
    \node[action, below=of unit] (collect) {Daten erheben};
    \node[action, below=of collect] (record) {Daten aufzeichnen};
    \node[action, below=of record] (analysis) {Daten auswerten};

    \node[fork, below left=of analysis] (merge1) {};
    \node[action, below=of merge1, xshift=-18mm] (comparison) {Ähnlichkeit prüfen /\\Synthese};
    \node[action, right=of comparison] (pub) {publizieren};

    \node[circle, draw, line width=0.8pt, minimum size=7mm, right=of pub] (finalouter) {};
    \node[circle, fill=black, minimum size=3.5mm, inner sep=0pt] at (finalouter.center) {};

    \node[rsepic, right=of collect, xshift=-8mm] (rse_collect) {\rsecycle};
    \node[rsepic, right=of record, xshift=-8mm] (rse_record) {\rsecycle};
    \node[rsepic, right=of analysis, xshift=-8mm] (rse_analysis) {\rsecycle};
    \node[rsepic, below=of simulation, yshift=+7mm, xshift=-4mm] (rse_simulation) {\rsecycle};

    \draw (start) -- (problem);
    \draw (problem) -- (theory);
    \draw (theory) -- (fork1);

    \draw (fork1.west) -| (concept.north);
    \draw (fork1.east) -| (form.north);

    \draw (concept) -- (merge_concept);
    \draw (merge_concept) -- (simulation);
    \draw (merge_concept) -- (unit);
    \draw (simulation.west) -- ++(-1,0) -- ++(0,-4.7) -- (merge1.west);

    \draw (form) -- (unit);
    \draw (unit) -- (collect);
    \draw (collect) -- (record);
    \draw (record) -- (analysis);
    \draw (analysis.south) |- (merge1);

    \draw (merge1.south) -- ++(0,-4mm) -| (comparison.north);
    \draw (comparison) -- (pub);
    \draw (pub) -- (finalouter);

    \end{tikzpicture}
\end{document}
